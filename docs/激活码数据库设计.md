# Roy 激活码与订阅数据库设计

## 数据库选型
推荐：PostgreSQL 15+（支持 JSONB、行级安全）

## 核心表结构

### 1. accounts（账号主表）
```sql
CREATE TABLE accounts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email VARCHAR(255) UNIQUE NOT NULL,
  username VARCHAR(50) UNIQUE NOT NULL,
  password_hash VARCHAR(255) NOT NULL, -- bcrypt
  plan VARCHAR(20) NOT NULL DEFAULT 'free', -- 'free' | 'pro'
  pro_expires_at TIMESTAMP NULL, -- Pro 过期时间
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_accounts_email ON accounts(email);
CREATE INDEX idx_accounts_username ON accounts(username);
CREATE INDEX idx_accounts_pro_expires ON accounts(pro_expires_at) WHERE pro_expires_at IS NOT NULL;
```

### 2. activation_codes（激活码表）
```sql
CREATE TABLE activation_codes (
  code VARCHAR(32) PRIMARY KEY, -- ROY-XXXX-XXXX-XXXX
  plan VARCHAR(20) NOT NULL, -- 'pro'
  duration_days INTEGER NOT NULL, -- 30 / 365
  max_uses INTEGER NOT NULL DEFAULT 1, -- 激活次数上限
  used_count INTEGER NOT NULL DEFAULT 0,
  batch_id UUID, -- 批次 ID（用于批量生成追踪）
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  expires_at TIMESTAMP NULL, -- 激活码本身的有效期（NULL = 永久）
  is_active BOOLEAN NOT NULL DEFAULT TRUE,
  CHECK (used_count <= max_uses)
);

CREATE INDEX idx_codes_batch ON activation_codes(batch_id);
CREATE INDEX idx_codes_expires ON activation_codes(expires_at) WHERE expires_at IS NOT NULL;
```

### 3. activations（激活记录表）
```sql
CREATE TABLE activations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  account_id UUID NOT NULL REFERENCES accounts(id) ON DELETE CASCADE,
  code VARCHAR(32) NOT NULL REFERENCES activation_codes(code),
  activated_at TIMESTAMP NOT NULL DEFAULT NOW(),
  duration_days INTEGER NOT NULL, -- 本次激活的天数
  expires_at TIMESTAMP NOT NULL, -- 本次激活的过期时间
  device_id VARCHAR(64) NOT NULL, -- 激活时的设备 ID
  device_name VARCHAR(255),
  UNIQUE(account_id, code) -- 同一个账号不能重复使用同一个激活码
);

CREATE INDEX idx_activations_account ON activations(account_id);
CREATE INDEX idx_activations_code ON activations(code);
CREATE INDEX idx_activations_expires ON activations(expires_at);
```

### 4. devices（设备绑定表）
```sql
CREATE TABLE devices (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  account_id UUID NOT NULL REFERENCES accounts(id) ON DELETE CASCADE,
  device_id VARCHAR(64) NOT NULL, -- 客户端生成的 UUID
  device_name VARCHAR(255) NOT NULL, -- hostname()
  device_fingerprint JSONB, -- 可选：CPU/MAC/硬盘序列号等硬件指纹
  last_seen TIMESTAMP NOT NULL DEFAULT NOW(),
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  UNIQUE(account_id, device_id)
);

CREATE INDEX idx_devices_account ON devices(account_id);
CREATE INDEX idx_devices_last_seen ON devices(last_seen);
```

### 5. subscriptions（订阅管理表，用于周期扣费）
```sql
CREATE TABLE subscriptions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  account_id UUID NOT NULL REFERENCES accounts(id) ON DELETE CASCADE,
  plan VARCHAR(20) NOT NULL, -- 'pro'
  status VARCHAR(20) NOT NULL, -- 'active' | 'canceled' | 'expired'
  billing_cycle VARCHAR(20) NOT NULL, -- 'monthly' | 'yearly'
  amount_cents INTEGER NOT NULL, -- 2900 / 39900
  currency VARCHAR(3) NOT NULL DEFAULT 'CNY',
  started_at TIMESTAMP NOT NULL,
  current_period_end TIMESTAMP NOT NULL,
  canceled_at TIMESTAMP NULL,
  payment_provider VARCHAR(50), -- 'wechat' | 'alipay' | 'stripe'
  payment_id VARCHAR(255), -- 第三方支付订单号
  created_at TIMESTAMP NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMP NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_subscriptions_account ON subscriptions(account_id);
CREATE INDEX idx_subscriptions_status ON subscriptions(status);
CREATE INDEX idx_subscriptions_period_end ON subscriptions(current_period_end);
```

## 业务逻辑

### 激活码使用流程
```sql
-- 1. 校验激活码
SELECT code, plan, duration_days, max_uses, used_count, expires_at
FROM activation_codes
WHERE code = $1
  AND is_active = TRUE
  AND used_count < max_uses
  AND (expires_at IS NULL OR expires_at > NOW());

-- 2. 创建激活记录
INSERT INTO activations (account_id, code, duration_days, expires_at, device_id, device_name)
VALUES ($1, $2, $3, NOW() + INTERVAL '$3 days', $4, $5);

-- 3. 更新激活码使用次数
UPDATE activation_codes
SET used_count = used_count + 1
WHERE code = $2;

-- 4. 更新账号 Pro 状态
UPDATE accounts
SET plan = 'pro',
    pro_expires_at = GREATEST(
      COALESCE(pro_expires_at, NOW()),  -- 已过期从现在算
      NOW()
    ) + INTERVAL '$3 days'
WHERE id = $1;
```

### 设备绑定检查
```sql
-- 获取账号当前绑定设备数
SELECT COUNT(*) FROM devices
WHERE account_id = $1;

-- 如果达到上限（free=2, pro=5），拒绝新设备
-- 否则 UPSERT
INSERT INTO devices (account_id, device_id, device_name, last_seen)
VALUES ($1, $2, $3, NOW())
ON CONFLICT (account_id, device_id)
DO UPDATE SET last_seen = NOW(), device_name = EXCLUDED.device_name;
```

### 生成权益凭证（Go/Node.js 示例）
```javascript
import { sign } from 'node:crypto'

function generateEntitlement(account, device) {
  const payload = {
    v: 1,
    sub: account.id,
    email: account.email,
    username: account.username,
    plan: account.plan,
    features: deriveFeatures(account),
    proExpiresAt: account.pro_expires_at?.getTime() ?? null,
    deviceId: device.device_id,
    iat: Date.now(),
    validUntil: Date.now() + 10 * 24 * 60 * 60 * 1000, // 10 天
  }
  
  const payloadB64 = Buffer.from(JSON.stringify(payload)).toString('base64url')
  const sig = sign(null, Buffer.from(payloadB64), PRIVATE_KEY).toString('base64url')
  
  return { payload: payloadB64, sig }
}

function deriveFeatures(account) {
  if (account.plan !== 'pro') return []
  return [
    'computer-control',
    'deepseek-mode',
    'mobile-remote-public',
    'cloud-sync',
    'unlimited-automation',
    'multi-agent',
    'long-memory',
  ]
}
```

## 密钥管理

### Ed25519 密钥对生成（一次性）
```bash
# 生成私钥
openssl genpkey -algorithm Ed25519 -out private.pem

# 提取公钥
openssl pkey -in private.pem -pubout -out public.pem

# 服务端保管 private.pem
# 客户端内置 public.pem
```

## 防篡改机制

| 攻击方式 | 防护措施 |
|---|---|
| 改 `hasProFeature()` 返回 `true` | 签名校验失败（公钥验证） |
| 伪造 entitlement | 没有私钥无法生成有效签名 |
| 复制凭证到另一台机器 | deviceId 不匹配，签名校验失败 |
| 修改 validUntil 延长有效期 | 改动后签名失效 |
| 离线长期使用 | validUntil 最多 10 天，超时后需联网刷新 |
